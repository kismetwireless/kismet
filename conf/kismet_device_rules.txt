# Kismet device rules
#
# Each rule names a condition on a device and what to do when a device matches it:
#
#   rule <name> {
#       when <condition>
#       description "<text>"    what the rule finds, shown with its tags in the web UI
#       tag "<text>"            tag matching devices (any number of tags)
#       color "#rrggbb"         highlight color for matching devices
#       view "<description>"    collect matching devices in the view rule-<name>; the view
#                               takes the rule's description when it has none of its own
#       alert                   raise a DEVICERULE alert when a device starts matching
#       live                    check on every packet as well as on the timer
#       exclude                 leave matching devices out of the 'default' view, which
#                               the web UI lists as Default
#       webhook "<endpoint>"    send matching devices to a webhook endpoint from
#                               kismet_webhooks.conf, optionally with { terms } below
#   }
#
# A webhook sends when a device starts matching, unless its terms say otherwise:
#
#   webhook "<endpoint>" {
#       trigger match           when a device starts matching (the default); again only
#                               after it stops matching and matches again
#       trigger new             when a device first seen after the rule loaded starts
#                               matching, once per device
#       trigger once            the first device to match, once; a reload keeps that
#                               unless the webhook's trigger changes
#       trigger every <time>    while a device matches, each time it's updated, at most
#                               once per <time> (10s or more)
#       trigger change <field>, <field> ... limit <time>
#                               when the fields change while a device matches, at most
#                               once per <time> per device (1s or more).  Fields are
#                               single values, such as signal or name.  Changes inside
#                               the limit are sent together when it's up, with the old
#                               and new values.  A limit is required.
#       for <time>              a device must keep matching this long before it's sent
#       stop                    also send when a device stops matching
#       rate <n>/<period> [burst <n>]
#                               most events from this webhook per period, such as 10/1m
#                               or 100/1h (default 10/1m)
#       batch <time> [max <n>]  hold events to send together: the first event opens a
#                               window, sent when <time> is up or <n> events are in it
#                               (default 100)
#       fields <field>, ...     the device fields to send, as in the REST api's fields
#                               (default: key, mac, name, phy, type, manuf, channel,
#                               frequency, first and last time, and signal)
#       fire_existing           also send for devices that already match when the rule
#                               loads; without it they're recorded and not sent
#   }
#
# The endpoint's own rate, queue and retry limits apply on top of these, along with
# webhook_global_rate; /webhooks/status.json shows what each webhook sent and held back.
#
# Conditions compare device fields to values:
#
#   Fields are Kismet field names, such as kismet.device.base.manuf, or a path through
#   nested fields joined with '/' (no spaces), such as
#   kismet.device.base.signal/kismet.common.signal.last_signal.  A path through a list
#   or map of records matches when any record matches.  Short names: mac, name, phy,
#   type, manuf, crypt, channel, frequency, packets, datasize, first_time, last_time,
#   mod_time, num_alerts, signal, min_signal, max_signal, seenby, location (the
#   average location) and last_location.
#
#   Numbers compare with == != < <= > >=, and take + - * / % (put spaces around '/'
#   for division).  Durations such as 30s, 5m, 2h and 1d are seconds; 'now' is the
#   current time and age(field) is now - field.  lat(field) and lon(field) are the
#   latitude and longitude of a location, such as lat(location); a device without
#   a location doesn't match.
#
#   Strings compare with == and !=, or with contains, icontains (any case),
#   startswith, endswith and matches (a PCRE regex), against a "string".  The
#   device type matches in any case: type == "wi-fi ap" matches "Wi-Fi AP".
#
#   MAC addresses are written aa:bb:cc:dd:ee:ff, with an optional mask as /24 or
#   /ff:ff:ff:00:00:00; a short address such as aa:bb:cc matches the prefix.
#
#   'x in [a, b, c]' and 'x not in [...]' test against a list of values.
#   exists(field) tests that a field is present, and count(field) counts entries.
#   has_tag("<name>") tests that the device has the user tag <name>, such as the
#   "baseline" tag the web UI sets from the device details.
#   Combine conditions with and, or, not, and parentheses.
#
#   A comparison with a field the device doesn't have is false.
#
# Other rules files can be read in with:
#
#   include "<file>"         a file, relative to this one
#   include "<dir>/*.rules"  every matching file, in natural order
#
# Plugins can ship rules files too, named in their manifest.conf with
# devicerules=<file>.  Plugin rules load after the rules files and before
# device_rule= lines in the config, so either can replace a plugin rule by name.
#
# Sorts order devices for the device views:
#
#   sort <name> {
#       by <value> [asc|desc], <value> [asc|desc] ...
#   }
#
#   A value is a field or a calculation on fields, a true/false condition (false
#   sorts first), or 'rule', the position of the first rule in this file the device
#   matches.  Later values break ties in earlier ones; devices missing a value sort
#   after the rest, in either direction.  Ask a view for a sort with rule_sort, as in
#   /devices/views/all/devices.json?rule_sort=<name>.
#
# Device views also take a condition and sort keys directly, as filter=<condition>
# and filter_sort=<value> [asc|desc], ...  In a filter, text("<s>") matches devices
# where any text field the request asks for contains s, in any case.
#
# Examples:
#
# rule close-apple {
#     description "Apple devices close by"
#     when manuf contains "Apple" and signal > -60
#     tag "close"
# }
#
# rule watchlist {
#     description "Devices on the watch list"
#     when mac in [00:11:22:33:44:55, aa:bb:cc/24]
#     tag "watchlist"
#     view "Devices on the watch list"
#     alert
#     live
# }
#
# rule hidden {
#     description "Devices tagged 'hide' in the web UI"
#     when has_tag("hide")
#     exclude
# }
#
# rule open-ap {
#     description "Open Wi-Fi access points"
#     when phy == "IEEE802.11" and type == "Wi-Fi AP" and crypt == "Open"
#     tag "open"
# }
#
# rule free-ssid {
#     description "Networks advertising free Wi-Fi"
#     when dot11.device/dot11.device.advertised_ssid_map/dot11.advertisedssid.ssid icontains "free"
#     tag "free wifi"
# }
#
# sort loudest {
#     by signal desc, packets desc
# }
#
# sort by-rule {
#     by rule, last_time desc
# }
#
# rule new-public-btle {
#     description "New BTLE devices with public addresses"
#     when phy == "BTLE" and not exists(btle.device.random_address_type)
#     webhook "ha" {
#         trigger new
#         batch 30s max 100
#         fields mac, name, signal
#     }
# }
#
# rule watchlist-signal {
#     when mac in [00:11:22:33:44:55]
#     webhook "ha" {
#         trigger change signal limit 30s
#         stop
#     }
# }
#
# rule gone-quiet {
#     description "Busy devices that have gone quiet"
#     when age(last_time) > 10m and packets > 1000
#     tag "quiet"
# }
#
# rule quad {
#     description "Devices seen inside a lat/lon box"
#     when lat(location) >= 10.0 and lat(location) <= 10.1 and lon(location) >= 20.0 and lon(location) <= 20.1
#     tag "quad"
#     view "Devices in the quad"
# }

# Baseline: devices you've marked as known with "Add to baseline" in the web UI's
# device details, which tags them "baseline", and every other device.

rule baseline {
    description "Devices in the baseline"
    when has_tag("baseline")
    view "Baseline devices"
}

rule unknown {
    description "Devices not in the baseline"
    when not has_tag("baseline")
    view "Unknown devices"
}
